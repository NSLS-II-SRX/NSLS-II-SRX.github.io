<p>&lt;script&gt;
function xas_step_estimate_time() {</p>
<blockquote>
<div><p>var erange_text = document.getElementById(‘xas_step_erange’).value || “0”;
var erange_list = erange_text.split(‘,’)</p>
<blockquote>
<div><p>.map(function(item) { return parseFloat(item.trim()); })
.filter(function(num) { return !isNaN(num); });</p>
</div></blockquote>
<p>var estep_text = document.getElementById(‘xas_step_estep’).value || “0”;
var estep_list = estep_text.split(‘,’)</p>
<blockquote>
<div><p>.map(function(item) { return parseFloat(item.trim()); })
.filter(function(num) { return !isNaN(num); });</p>
</div></blockquote>
<p>var dwell = Math.max(0, parseFloat(document.getElementById(‘xas_step_dwell’).value) || 0);</p>
<p>// Get results first to display errors
var res = document.getElementById(‘xas_step_res’);
if (!res) return; // Guard clause if element doesn’t exist</p>
<p>// CHECK 1: Ensure steps list length is exactly 1 less than erange list length
if (erange_list.length !== estep_list.length + 1) {</p>
<blockquote>
<div><p>res.innerText = “Error: erange must be exactly 1 longer than estep”;
res.style.color = “red”; // Optional: Makes error text pop out
return; // Stop the script immediately</p>
</div></blockquote>
<p>}</p>
<p>// CHECK 2: Ensure erange numbers are sequential (no duplicates, no direction changes)
for (var j = 0; j &lt; erange_list.length - 2; j++) {</p>
<blockquote>
<div><p>var first_diff = erange_list[j+1] - erange_list[j];
var second_diff = erange_list[j+2] - erange_list[j+1];</p>
<p>// If a difference is 0, they entered a duplicate number.
// If multiplying the differences yields a negative number, the scan changed direction.
if (first_diff === 0 || second_diff === 0 || (first_diff * second_diff) &lt; 0) {</p>
<blockquote>
<div><p>res.innerText = “Error: erange values must be sequential”;
res.style.color = “red”;
return;</p>
</div></blockquote>
<p>}</p>
</div></blockquote>
<p>}
res.style.color = “”;</p>
<p>// Get total points by looping
var total_points = 0;</p>
<p>// Loop through each segment (defined by steps list)
for (var i = 0; i &lt; estep_list.length; i++) {</p>
<blockquote>
<div><p>// Ensure we have a matching pair of start/end points for this step size
if (i + 1 &lt; erange_list.length) {</p>
<blockquote>
<div><p>var start = erange_list[i];
var end = erange_list[i + 1];
var step = estep_list[i];</p>
<dl class="simple">
<dt>if (step &gt; 0) {</dt><dd><p>// Points in segment = Math.abs(end - start) / step
// We use Math.round to fix floating-point math issues (e.g., 0.1 + 0.2 = 0.30000004)
var segment_steps = Math.round(Math.abs(end - start) / step);
total_points += segment_steps;</p>
</dd>
</dl>
<p>}</p>
</div></blockquote>
<p>}</p>
</div></blockquote>
<p>}</p>
<p>// Add 1 for the very first starting point if any points were created
if (total_points &gt; 0) {</p>
<blockquote>
<div><p>total_points += 1;</p>
</div></blockquote>
<p>}
// —————————————————–</p>
<p>var res_sec = 10 + (total_points * (dwell + 4));</p>
<p>// Dwell issue
if (dwell == 0) {</p>
<blockquote>
<div><p>res_sec = 0</p>
</div></blockquote>
<p>}</p>
<p>res.innerText = formatTimeFromSeconds(res_sec);</p>
</div></blockquote>
<p>}
&lt;/script&gt;</p>
